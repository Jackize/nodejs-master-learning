2. Service + Controller + Module

src/catalog/product.types.ts 
export type Product = {
  id: string;
  name: string;
  price: number;
  stock: number;
};

/** Class — không dùng type-only với @Body() khi emitDecoratorMetadata bật */
export class CreateProductInput {
  name!: string;
  price!: number;
  stock!: number;
}
src/catalog/products.service.ts 
import { Injectable } from '@nestjs/common';
import { randomUUID } from 'crypto';
import { CreateProductInput, Product } from './product.types';

@Injectable()
export class ProductsService {
  // RAM tạm — Bài 05 thay Mongo
  private readonly products = new Map<string, Product>();

  findAll(): Product[] {
    return [...this.products.values()];
  }

  create(input: CreateProductInput): Product {
    const product: Product = {
      id: randomUUID(),
      name: input.name,
      price: input.price,
      stock: input.stock,
    };
    this.products.set(product.id, product);
    return product;
  }

  findById(id: string): Product | undefined {
    return this.products.get(id);
  }
}
src/catalog/products.controller.ts 
import { Body, Controller, Get, Post } from '@nestjs/common';
import { CreateProductInput } from './product.types';
import { ProductsService } from './products.service';

@Controller('catalog/products')
export class ProductsController {
  // DI: Nest tạo ProductsService vì CatalogModule khai báo providers
  constructor(private readonly productsService: ProductsService) {}

  @Get()
  list() {
    return this.productsService.findAll();
  }

  @Post()
  create(@Body() body: CreateProductInput) {
    return this.productsService.create(body);
  }
}
src/catalog/catalog.module.ts 
import { Module } from '@nestjs/common';
import { ProductsController } from './products.controller';
import { ProductsService } from './products.service';

@Module({
  controllers: [ProductsController],
  providers: [ProductsService],
  // Shared module: cùng 1 instance khi Cart import CatalogModule
  exports: [ProductsService],
})
export class CatalogModule {}
Export ≠ đăng ký lại provider

Theo docs: nếu khai báo ProductsService trong nhiều module thay vì export/import, mỗi module nhận instance riêng — state Map sẽ lệch. Đó là lý do Cart sau này imports: [CatalogModule].

4. Unit test service + sanity HTTP

src/catalog/products.service.spec.ts 
import { ProductsService } from './products.service';

describe('ProductsService', () => {
  let service: ProductsService;
  beforeEach(() => { service = new ProductsService(); });

  it('ban đầu danh sách rỗng', () => {
    expect(service.findAll()).toEqual([]);
  });

  it('create rồi findAll có 1 sản phẩm', () => {
    const created = service.create({ name: 'Áo thun', price: 199000, stock: 10 });
    expect(created.id).toBeDefined();
    expect(service.findAll()).toHaveLength(1);
    expect(service.findById(created.id)).toEqual(created);
  });
});
